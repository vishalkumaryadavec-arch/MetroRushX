function checkCollision(o) {
if (o.collected) return false;
if (o.z > 0.88) return false;
if (Math.abs(o.lane - state.lane) > 0.22) return false;
if (o.type === "coin") return true;

o.type === "shield" ||
o.type === "magnet" ||
o.type === "boost"

return true;
}
if (o.type === "obstacle" && state.playerY > 55) return false;
if (o.type === "train" && state.sliding) return false;

}

/* =========================================================
GAME UPDATE
========================================================= */
function update(dt) {
if (!state.running || state.paused) return;
state.runTime += dt;
state.distance += state.speed * dt * 4;
state.score += state.speed * dt * state.combo;
missionProgress.distance = state.distance;
missionProgress.score = state.score;
missionProgress.time = state.runTime;
state.lane +=
(state.targetLane - state.lane) * Math.min(dt * 12, 1);
state.playerY += state.velocityY * dt;
state.velocityY -= 36 * dt;
if (state.playerY < 0) {
state.playerY = 0;
state.velocityY = 0;
}
state.speed += dt * 0.035;
state.spawnTimer -= dt;
state.coinTimer -= dt;
if (state.spawnTimer <= 0) {
spawnPattern();
state.spawnTimer =
Math.max(0.35, 0.85 - state.distance / 15000) +
Math.random() * 0.35;
}
if (state.coinTimer <= 0) {
spawnObject("coin");
state.coinTimer = 0.55;
}
for (const o of state.objects) {
o.z -= state.speed * dt * 0.055;
if (
state.magnet &&
o.type === "coin" &&
Math.abs(o.lane - state.lane) < 0.5 &&
o.z < 0.8
) {
collect(o);
continue;
}
if (checkCollision(o)) {
if (

o.type === "coin" ||
o.type === "shield" ||
o.type === "magnet" ||
o.type === "boost"
) {
collect(o);
} else {
hit(o);
}
}
}
state.objects = state.objects.filter(
(o) => o.z > -0.15 && !o.collected
);
state.combo = Math.min(
5,
1 + Math.floor(state.distance / 1000) * 0.5
);
updateHUD();
}

<div style="padding:12px;margin:8px 0;background:rgba(255,255,255,.08);border-radius:12px">
<b>${mission.text}</b>
<div>${Math.floor(progress)} / ${mission.target}</div>
<div style="height:8px;background:#333;border-radius:8px;overflow:hidden">
<div style="width:${percent}%;height:100%;background:#31d67b"></div>
</div>
</div>
`;
})
.join("");
}
/* =========================================================
DAILY
========================================================= */
function renderDaily() {
const container = $("dailyContent") || $("dailyList");
if (!container) return;
const today = new Date().toISOString().slice(0, 10);
const key = "metroRushDaily";
let daily = JSON.parse(localStorage.getItem(key) || "null");
if (!daily || daily.date !== today) {
daily = {
date: today,
claimed: false,
target: 10,
progress: 0,
reward: 100
};
localStorage.setItem(key, JSON.stringify(daily));
}
container.innerHTML = `
<div style="padding:14px">
<h3>Daily Challenge</h3>
<p>Collect ${daily.target} coins.</p>
<p>Progress: ${daily.progress}/${daily.target}</p>
<button id="dailyClaim" ${daily.claimed ? "disabled" : ""}>
${daily.claimed ? "Claimed" : "Claim Reward: " + daily.reward + " coins"}
</button>
</div>
`;
$("dailyClaim")?.addEventListener("click", () => {
if (daily.claimed) return;
if (state.coins < daily.target && !state.running) {
toast("Collect coins during a run first!");
return;
}
daily.claimed = true;
state.bank += daily.reward;
saveGame();
renderDaily();
updateHUD();
toast("Daily reward claimed!");
});
}
